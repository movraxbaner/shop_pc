-- ---------------------------------------------------------------------
-- 2. ТАБЛИЦЫ
-- ---------------------------------------------------------------------

-- Роли сотрудников
CREATE TABLE Roles (
    Id       INT IDENTITY(1,1) PRIMARY KEY,
    RoleName NVARCHAR(50) NOT NULL UNIQUE
);

-- Категории комплектующих
CREATE TABLE Categories (
    Id   INT IDENTITY(1,1) PRIMARY KEY,
    Name NVARCHAR(100) NOT NULL UNIQUE
);

-- Пользователи системы (сотрудники)
CREATE TABLE Users (
    Id           INT IDENTITY(1,1) PRIMARY KEY,
    Login        NVARCHAR(50)  NOT NULL UNIQUE,
    PasswordHash NVARCHAR(255) NOT NULL,           -- SHA-256 в hex (см. комментарий ниже)
    FullName     NVARCHAR(150) NOT NULL,
    RoleId       INT NOT NULL,
    IsActive     BIT NOT NULL DEFAULT 1,           -- админ может заблокировать, не удаляя
    CONSTRAINT FK_Users_Roles FOREIGN KEY (RoleId) REFERENCES Roles(Id)
);

-- Товары и комплектующие
CREATE TABLE Products (
    Id             INT IDENTITY(1,1) PRIMARY KEY,
    Sku            NVARCHAR(50)  NOT NULL UNIQUE,  -- артикул
    Name           NVARCHAR(255) NOT NULL,
    CategoryId     INT NOT NULL,
    Price          DECIMAL(10,2) NOT NULL CHECK (Price >= 0),
    StockQuantity  INT NOT NULL DEFAULT 0 CHECK (StockQuantity >= 0),
    Specifications NVARCHAR(MAX) NULL,
    IsActive       BIT NOT NULL DEFAULT 1,         -- «удаление» товара = IsActive = 0
    RowVersion     ROWVERSION,                     -- защита от параллельных правок (EF Core)
    CONSTRAINT FK_Products_Categories FOREIGN KEY (CategoryId) REFERENCES Categories(Id)
);

-- Заказы / продажи
CREATE TABLE Orders (
    Id          INT IDENTITY(1,1) PRIMARY KEY,
    OrderDate   DATETIME NOT NULL DEFAULT GETDATE(),
    UserId      INT NOT NULL,                      -- сотрудник, оформивший заказ
    TotalAmount DECIMAL(10,2) NOT NULL DEFAULT 0 CHECK (TotalAmount >= 0),
    Status      NVARCHAR(50) NOT NULL DEFAULT N'Новый',
    CONSTRAINT FK_Orders_Users FOREIGN KEY (UserId) REFERENCES Users(Id),
    CONSTRAINT CK_Orders_Status CHECK (Status IN (N'Новый', N'Проведён', N'Отменён'))
);

-- Позиции заказа
CREATE TABLE OrderItems (
    Id        INT IDENTITY(1,1) PRIMARY KEY,
    OrderId   INT NOT NULL,
    ProductId INT NOT NULL,
    Quantity  INT NOT NULL CHECK (Quantity > 0),
    UnitPrice DECIMAL(10,2) NOT NULL CHECK (UnitPrice >= 0),  -- цена на момент продажи
    CONSTRAINT FK_OrderItems_Orders   FOREIGN KEY (OrderId)   REFERENCES Orders(Id) ON DELETE CASCADE,
    CONSTRAINT FK_OrderItems_Products FOREIGN KEY (ProductId) REFERENCES Products(Id),
    CONSTRAINT UQ_OrderItems_Order_Product UNIQUE (OrderId, ProductId)  -- один товар = одна строка
);

-- Списание товара
CREATE TABLE WriteOffs (
    Id           INT IDENTITY(1,1) PRIMARY KEY,
    ProductId    INT NOT NULL,
    Quantity     INT NOT NULL CHECK (Quantity > 0),
    Reason       NVARCHAR(255) NOT NULL,
    WriteOffDate DATETIME NOT NULL DEFAULT GETDATE(),
    UserId       INT NOT NULL,                     -- кто списал
    CONSTRAINT FK_WriteOffs_Products FOREIGN KEY (ProductId) REFERENCES Products(Id),
    CONSTRAINT FK_WriteOffs_Users    FOREIGN KEY (UserId)    REFERENCES Users(Id)
);


-- ---------------------------------------------------------------------
-- 3. ИНДЕКСЫ (UNIQUE-поля индексируются автоматически)
-- ---------------------------------------------------------------------
CREATE INDEX IX_Products_Name       ON Products(Name);
CREATE INDEX IX_Products_CategoryId ON Products(CategoryId);
CREATE INDEX IX_Orders_OrderDate    ON Orders(OrderDate) INCLUDE (Status);
CREATE INDEX IX_Orders_UserId       ON Orders(UserId);
CREATE INDEX IX_OrderItems_ProductId ON OrderItems(ProductId);
CREATE INDEX IX_WriteOffs_ProductId ON WriteOffs(ProductId);
GO

-- ---------------------------------------------------------------------
-- 5. ФУНКЦИИ И ПРЕДСТАВЛЕНИЯ ДЛЯ ОТЧЁТОВ
--    Период: [From, To) — конец НЕ включается
--    (для октября: '2026-10-01' и '2026-11-01')
-- ---------------------------------------------------------------------

-- Продажи по дням за период
CREATE FUNCTION dbo.fn_SalesByDay (@From DATETIME, @To DATETIME)
RETURNS TABLE AS RETURN (
    SELECT CAST(o.OrderDate AS DATE)          AS SaleDate,
           COUNT(DISTINCT o.Id)               AS OrdersCount,
           SUM(oi.Quantity)                   AS ItemsSold,
           SUM(oi.Quantity * oi.UnitPrice)    AS Revenue
    FROM Orders o
    JOIN OrderItems oi ON oi.OrderId = o.Id
    WHERE o.OrderDate >= @From AND o.OrderDate < @To
      AND o.Status = N'Проведён'
    GROUP BY CAST(o.OrderDate AS DATE)
);
GO

-- ТОП-5 самых продаваемых товаров за период
CREATE FUNCTION dbo.fn_TopProducts (@From DATETIME, @To DATETIME)
RETURNS TABLE AS RETURN (
    SELECT TOP 5
           p.Id, p.Sku, p.Name,
           SUM(oi.Quantity)                AS TotalQty,
           SUM(oi.Quantity * oi.UnitPrice) AS Revenue
    FROM OrderItems oi
    JOIN Orders o   ON o.Id = oi.OrderId
    JOIN Products p ON p.Id = oi.ProductId
    WHERE o.OrderDate >= @From AND o.OrderDate < @To
      AND o.Status = N'Проведён'
    GROUP BY p.Id, p.Sku, p.Name
    ORDER BY SUM(oi.Quantity) DESC
);
GO

-- Товары, которые заканчиваются (остаток меньше 3)
CREATE VIEW dbo.vw_LowStock AS
SELECT p.Id, p.Sku, p.Name, c.Name AS Category, p.StockQuantity
FROM Products p
JOIN Categories c ON c.Id = p.CategoryId
WHERE p.IsActive = 1 AND p.StockQuantity < 3;
GO
